# 液相色谱仪控制系统开发手册 V3.0

## 1. 产品概述

### 1.1 产品定位
液相色谱仪控制系统是一款面向分析实验室的专业自动化控制软件，专注于提供高效、精准、智能的色谱分离和纯化解决方案。系统采用B/S架构，支持远程操作和实时监控，实现从样品进样到数据分析的全流程自动化。

### 1.2 核心价值
- **自动化程度高**：减少80%以上的人工操作，支持24/7无人值守运行
- **分析精度高**：智能峰检测算法，准确率>99%，支持肩峰识别
- **操作效率高**：支持批量样品连续分析，智能排程优化
- **数据可追溯**：完整的审计追踪和数据管理，符合GMP/GLP规范
- **维护成本低**：模块化设计，故障自诊断，远程维护支持

```python
class HighPressurePump:
    def __init__(self):
        self.solvent_a_port = 1
        self.solvent_b_port = 2
        
    def set_flow_rate(self, rate: float):
        """设置流速 (0.01-10.00 mL/min)"""
        
    def set_solvent_ports(self, solvent_a_port: int, solvent_b_port: int):
        """设置溶液孔位，1-4表示4个孔"""
        if not (1 <= solvent_a_port <= 4 and 1 <= solvent_b_port <= 4):
            raise ValueError("孔位必须在1-4之间")
        if solvent_a_port == solvent_b_port:
            raise ValueError("两个溶液不能使用同一个孔")
        self.solvent_a_port = solvent_a_port
        self.solvent_b_port = solvent_b_port
        
    def set_port_ratio(self, ratio_a: float, ratio_b: float):
        """设置两个溶液的比例，按set_solvent_ports设置的顺序"""
        if abs(ratio_a + ratio_b - 100) > 0.01:
            raise ValueError("两个比例之和必须为100")
        if ratio_a < 0 or ratio_b < 0:
            raise ValueError("比例不能为负数")
        
        # 根据孔位设置实际配置
        port_config = [0, 0, 0, 0]
        port_config[self.solvent_a_port - 1] = ratio_a
        port_config[self.solvent_b_port - 1] = ratio_b
        self.port_config = port_config
            
    def start_pump(self):
        """开始泵运行"""
        
    def set_gradient(self, gradient_table: list):
        """设置梯度程序"""
        
    def emergency_stop(self):
        """紧急停止"""
```

```python
class BubbleSensor:
    def __init__(self, sensor_id: int, gpio_pin: int):
        self.sensor_id = sensor_id
        self.gpio_pin = gpio_pin
        
    def has_bubble(self) -> bool:
        """检测是否有气泡，True=有气泡（无液体），False=有液体"""
        return GPIO.input(self.gpio_pin) == 1
        
    def get_sensor_status(self) -> dict:
        """获取传感器状态"""
        return {
            'sensor_id': self.sensor_id,
            'has_liquid': not self.has_bubble(),
            'status': 'ok' if not self.has_bubble() else 'no_liquid'
        }
```

#### 2.1.6 传感器状态管理器
**文件路径**: `host/sensor_manager.py`

```python
class SensorManager:
    def __init__(self, redis_client):
        self.redis_client = redis_client
        self.pressure_sensor = PressureSensor()
        self.bubble_sensors = {
            1: BubbleSensor(1, GPIO_1),
            2: BubbleSensor(2, GPIO_2), 
            3: BubbleSensor(3, GPIO_3),
            4: BubbleSensor(4, GPIO_4)
        }
        
    def check_pump_ports_ready(self, port_a: int, port_b: int) -> dict:
        """检查泵选择的孔位对应管路是否有液体"""
        result = {'ready': True, 'errors': []}
        
        # 检查孔A对应的管路
        if self.bubble_sensors[port_a].has_bubble():
            error = {
                'device': f'bubble_sensor_{port_a}',
                'alarm_type': 'no_liquid',
                'message': f'孔{port_a}对应管路无液体',
                'severity': 'high'
            }
            result['errors'].append(error)
            result['ready'] = False
            
        # 检查孔B对应的管路  
        if self.bubble_sensors[port_b].has_bubble():
            error = {
                'device': f'bubble_sensor_{port_b}',
                'alarm_type': 'no_liquid', 
                'message': f'孔{port_b}对应管路无液体',
                'severity': 'high'
            }
            result['errors'].append(error)
            result['ready'] = False
            
        # 发布错误到Redis
        for error in result['errors']:
            self.redis_client.publish('device_alarms', json.dumps(error))
            
        return result
        
    def check_all_sensors(self) -> dict:
        """检查所有传感器状态"""
        status = {
            'pressure_alarm': self.pressure_sensor.check_pressure_alarm(),
            'bubble_sensors': {},
            'overall_status': 'ok'
        }
        
        # 检查所有气泡传感器
        for sensor_id, sensor in self.bubble_sensors.items():
            sensor_status = sensor.get_sensor_status()
            status['bubble_sensors'][sensor_id] = sensor_status
            if sensor_status['status'] != 'ok':
                status['overall_status'] = 'warning'
                
        if status['pressure_alarm']:
            status['overall_status'] = 'alarm'
            
        return status
```

```python
class TubeRackHardwareController:
    def __init__(self):
        self.valve_controller = MultiValveController()
        self.current_position = 1
        
    def switch_to_tube(self, tube_number: int) -> bool:
        """切换到指定试管位置 - 纯硬件操作"""
        try:
            # 根据试管号计算需要切换的阀门和端口
            valve_config = self.calculate_valve_position(tube_number)
            
            # 执行具体的阀门切换
            success = self.valve_controller.switch_valve(
                valve_config['valve_num'], 
                valve_config['port_num']
            )
            
            if success:
                self.current_position = tube_number
                
            return success
            
        except Exception as e:
            print(f"阀门切换失败: {e}")
            return False
            
    def calculate_valve_position(self, tube_number: int) -> dict:
        """计算试管号对应的阀门配置"""
        if 1 <= tube_number <= 10:
            return {'valve_num': 2, 'port_num': tube_number}
        elif 11 <= tube_number <= 20:
            return {'valve_num': 3, 'port_num': tube_number - 10}
        elif 21 <= tube_number <= 30:
            return {'valve_num': 8, 'port_num': tube_number - 20}
        elif 31 <= tube_number <= 40:
            return {'valve_num': 9, 'port_num': tube_number - 30}
        else:
            raise ValueError(f"试管号 {tube_number} 超出范围")
            
    def switch_to_waste(self, waste_position: int = 1) -> bool:
        """切换到废液管路 - 纯硬件操作"""
        return self.valve_controller.switch_valve(6, waste_position)  # 电6切换到废液
    
    def get_current_configuration(self) -> dict:
        """获取当前阀门配置"""
        return {
            'current_tube_position': self.current_position,
            'valve_states': self.valve_controller.get_all_valve_states(),
            'timestamp': time.time()
        }
```

```python
        self.is_collecting = True  # True=收集到试管, False=排到废液
        self.switch_to_waste = False  # 切换到废液标志
        self.switch_to_collect = False  # 切换到收集标志
        self.hardware_controller = hardware_controller  # 硬件控制接口
        
    def trigger_manual_switch(self):
        """手动触发切换到下一管"""
        self.manual_trigger = True
        
    def switch_to_waste_mode(self):
        """切换到废液模式"""
        self.switch_to_waste = True
        
    def switch_to_collect_mode(self):
        """切换到收集模式"""
        self.switch_to_collect = True
        
    def set_target_volume(self, volume: float):
        """设置每个试管的目标收集体积"""
        self.target_volume_per_tube = volume
        
    def update_collected_volume(self, time_elapsed: float) -> dict:
        """更新已收集体积，返回切换状态"""
        result = {
            'switch_tube': False,
            'switch_to_waste': False,
            'switch_to_collect': False,
            'action_taken': None
        }
        
        # 处理废液/收集模式切换
        if self.switch_to_waste:
            self.is_collecting = False
            success = self.hardware_controller.switch_to_waste()
            result['switch_to_waste'] = success
            result['action_taken'] = 'switched_to_waste'
            self.switch_to_waste = False
            return result
            
        if self.switch_to_collect:
            self.is_collecting = True
            success = self.hardware_controller.switch_to_tube(self.current_tube)
            result['switch_to_collect'] = success
            result['action_taken'] = 'switched_to_collect'
            self.switch_to_collect = False
            return result
        
        # 只有在收集模式下才积分体积
        if self.is_collecting:
            # 根据流速和时间计算体积增量
            volume_increment = self.flow_rate * (time_elapsed / 60.0)  # 转换为分钟
            self.current_volume += volume_increment
            
            # 检查是否需要切换试管（体积达到目标 或 手动触发）
            should_switch = False
            
            # 体积达到目标自动切换
            if self.current_volume >= self.target_volume_per_tube:
                should_switch = True
                
            # 手动触发切换
            if self.manual_trigger:
                should_switch = True
                self.manual_trigger = False  # 重置手动触发标志
                
            if should_switch:
                next_tube = self.switch_to_next_tube()
                success = self.hardware_controller.switch_to_tube(next_tube)
                result['switch_tube'] = success
                result['action_taken'] = f'switched_to_tube_{next_tube}'
                
        return result
        
    def switch_to_next_tube(self) -> int:
        """切换到下一根试管"""
        # 记录当前试管的最终体积
        self.tube_status[self.current_tube] = {
            'final_volume': self.current_volume,
            'status': 'completed',
            'timestamp': time.time()
        }
        
        # 切换到下一根试管
        self.current_tube += 1
        self.current_volume = 0.0
        
        return self.current_tube
    
    def get_current_tube_info(self) -> dict:
        """获取当前试管信息"""
        return {
            'tube_number': self.current_tube,
            'collected_volume': self.current_volume,
            'target_volume': self.target_volume_per_tube,
            'progress_percent': (self.current_volume / self.target_volume_per_tube) * 100,
            'is_collecting': self.is_collecting,  # 当前是否在收集模式
            'collection_status': 'collecting' if self.is_collecting else 'waste'
        }
        
    def get_next_available_tube(self) -> int:
        """获取下一个可用试管"""
        return self.current_tube + 1
        
    def mark_tube_used(self, tube_number: int, final_volume: float):
        """标记试管已使用"""
        self.tube_status[tube_number] = {
            'final_volume': final_volume,
            'status': 'used',
            'timestamp': time.time()
        }
```

```python
class CollectionStrategy:
    """业务逻辑层 - 智能收集策略"""
    def __init__(self, mode: str, volume_manager: TubeVolumeManager):
        self.mode = mode  # "time", "volume", "peak", "manual"
        self.volume_manager = volume_manager
        
        # 模式相关参数
        self.time_interval = 300.0  # 秒，时间驱动的切换间隔
        self.volume_limit = 10.0    # ml，体积驱动的切换阈值
        
    def should_switch_tube(self, current_state: dict) -> bool:
        """判断是否切换试管 - 业务决策"""
        if self.mode == "peak":
            # 峰驱动收集
            if current_state["peak_detected"]:
                self.volume_manager.switch_to_collect_mode()
                return True
            if current_state["peak_ended"]:
                self.volume_manager.switch_to_waste_mode()
                return True
                
        elif self.mode == "time":
            # 时间驱动收集
            if current_state["elapsed_time"] >= self.time_interval:
                self.volume_manager.trigger_manual_switch()  # 触发切换
                return True
                
        elif self.mode == "volume":
            # 体积驱动收集
            if current_state["collected_volume"] >= self.volume_limit:
                self.volume_manager.trigger_manual_switch()  # 触发切换
                return True
                
        return False
        
    def make_collection_decision(self, uv_data: dict, flow_state: dict) -> dict:
        """收集决策制定 - 业务逻辑"""
        decision = {
            'collect': False,
            'waste': False,
            'switch_tube': False,
            'reason': ''
        }
        
        # 基于UV信号判断
        if uv_data['peak_detected']:
            decision['collect'] = True
            decision['reason'] = 'peak_detected'
        elif uv_data['baseline_period']:
            decision['waste'] = True
            decision['reason'] = 'baseline_period'
            
        # 基于体积判断
        if flow_state['tube_volume'] >= self.volume_manager.target_volume_per_tube:
            decision['switch_tube'] = True
            decision['reason'] += '_volume_full'
            
        return decision
```

```python
        """

        :param volumes: 格式如 {'A': 20, 'B': 20, 'C': 15, 'D': 10} (单位: mL)
        """
        for solvent, volume in volumes.items():
            if solvent in self.solvents:
                self.solvents[solvent]['initial_volume'] = volume
                self.solvents[solvent]['current_volume'] = volume
                self.solvents[solvent]['used_volume'] = 0
                
        print("原液体积初始化完成:")
        for solvent, data in self.solvents.items():
            if data['initial_volume'] > 0:
                print(f"  {solvent}液: {data['initial_volume']} mL")
    
    def update_solvent_usage(self, solvent_ratios: dict, time_elapsed: float):
        """
        根据溶剂比例和时间更新原液使用量
        
        :param solvent_ratios: 格式如 {'A': 80, 'B': 20, 'C': 0, 'D': 0} (百分比)
        :param time_elapsed: 经过的时间(秒)
        """
        # 计算总消耗体积
        total_volume_consumed = self.current_flow_rate * (time_elapsed / 60.0)  # 转换为分钟
        
        # 按比例分配到各个溶剂
        for solvent, ratio in solvent_ratios.items():
            if solvent in self.solvents and ratio > 0:
                consumed = total_volume_consumed * (ratio / 100.0)
                self.solvents[solvent]['used_volume'] += consumed
                self.solvents[solvent]['current_volume'] = max(0, 
                    self.solvents[solvent]['initial_volume'] - self.solvents[solvent]['used_volume'])
        
        self.last_update_time = time.time()
    
    def set_flow_rate(self, flow_rate: float):
        """设置当前流速"""
        self.current_flow_rate = flow_rate
    
    def check_sufficient_volume_for_experiment(self, experiment_time_minutes: float, 
                                             expected_ratios: dict) -> dict:
        """
        检查原液是否足够支持整个实验
        
        :param experiment_time_minutes: 实验时间(分钟)
        :param expected_ratios: 预期平均溶剂比例 {'A': 70, 'B': 30, ...}
        :return: 检查结果
        """
        result = {
            'sufficient': True,
            'warnings': [],
            'estimated_usage': {},
            'remaining_after_experiment': {}
        }
        
        # 计算预期总消耗量
        total_volume_needed = self.current_flow_rate * experiment_time_minutes
        
        for solvent, ratio in expected_ratios.items():
            if solvent in self.solvents and ratio > 0:
                needed_volume = total_volume_needed * (ratio / 100.0)
                current_available = self.solvents[solvent]['current_volume']
                
                result['estimated_usage'][solvent] = needed_volume
                result['remaining_after_experiment'][solvent] = current_available - needed_volume
                
                if needed_volume > current_available:
                    result['sufficient'] = False
                    shortage = needed_volume - current_available
                    result['warnings'].append({
                        'solvent': solvent,
                        'type': 'insufficient_volume',
                        'needed': needed_volume,
                        'available': current_available,
                        'shortage': shortage,
                        'message': f'{solvent}液不足！需要 {needed_volume:.1f}mL，可用 {current_available:.1f}mL，缺少 {shortage:.1f}mL'
                    })
                elif current_available - needed_volume < 2:  # 剩余少于2mL时预警
                    result['warnings'].append({
                        'solvent': solvent,
                        'type': 'low_volume_warning',
                        'message': f'{solvent}液将消耗殆尽，实验后剩余 {current_available - needed_volume:.1f}mL'
                    })
        
        return result
    
    def get_current_status(self) -> dict:
        """获取当前原液状态"""
        status = {}
        for solvent, data in self.solvents.items():
            if data['initial_volume'] > 0:
                usage_percent = (data['used_volume'] / data['initial_volume']) * 100
                status[solvent] = {
                    'initial': data['initial_volume'],
                    'current': data['current_volume'],
                    'used': data['used_volume'],
                    'usage_percent': usage_percent,
                    'is_low': data['current_volume'] < 3,  # 少于3mL时标记为低液位
                    'is_empty': data['current_volume'] <= 0
                }
        return status
    
    def handle_bubble_sensor_alarm(self, sensor_id: int) -> dict:
        """
        处理气泡传感器报警（检测到无液体）
        
        :param sensor_id: 传感器ID (1-4对应气1-气4)
        :return: 处理结果
        """
        # 传感器与溶剂的对应关系（需要根据实际连接配置）
        sensor_solvent_map = {1: 'A', 2: 'B', 3: 'C', 4: 'D'}
        
        if sensor_id in sensor_solvent_map:
            solvent = sensor_solvent_map[sensor_id]
            
            # 标记该溶剂为空
            self.solvents[solvent]['current_volume'] = 0
            
            alarm_data = {
                'alarm_type': 'solvent_empty',
                'sensor_id': sensor_id,
                'solvent': solvent,
                'message': f'气{sensor_id}传感器检测到{solvent}液耗尽！实验已暂停',
                'severity': 'critical',
                'action_required': f'请补充{solvent}液后重新启动实验',
                'experiment_action': 'pause'
            }
            
            # 发送报警
            send_frontend_alarm(alarm_data)
            
            return alarm_data
        
        return {'error': f'未知传感器ID: {sensor_id}'}
```

```python
class WasteManager:
    """废液管理系统"""
    
    def __init__(self):
        self.waste_container_capacity = 0  # 废液桶容积(mL)
        self.current_waste_volume = 0      # 当前废液体积(mL)
        self.waste_log = []                # 废液记录
        
    def initialize_waste_container(self, capacity: float):
        """
        初始化废液桶容积
        
        :param capacity: 废液桶容积(mL)
        """
        self.waste_container_capacity = capacity
        self.current_waste_volume = 0
        print(f"废液桶容积初始化: {capacity} mL")
    
    def calculate_tube_waste_volume(self, tubes_to_waste: list) -> dict:
        """
        计算将要废弃的试管总体积
        
        :param tubes_to_waste: 要废弃的试管列表，格式如 [{'tube_id': 1, 'volume': 10.5}, ...]
        :return: 计算结果
        """
        total_volume = sum(tube['volume'] for tube in tubes_to_waste)
        
        result = {
            'total_waste_volume': total_volume,
            'tube_count': len(tubes_to_waste),
            'current_waste_volume': self.current_waste_volume,
            'available_capacity': self.waste_container_capacity - self.current_waste_volume,
            'will_overflow': False,
            'overflow_volume': 0
        }
        
        # 检查是否会溢出
        if self.current_waste_volume + total_volume > self.waste_container_capacity:
            result['will_overflow'] = True
            result['overflow_volume'] = (self.current_waste_volume + total_volume) - self.waste_container_capacity
        
        return result
    
    def check_waste_capacity_before_disposal(self, tubes_to_waste: list) -> dict:
        """
        废弃前检查废液桶容量
        
        :param tubes_to_waste: 要废弃的试管列表
        :return: 检查结果和建议
        """
        calculation = self.calculate_tube_waste_volume(tubes_to_waste)
        
        result = {
            'can_proceed': True,
            'warnings': [],
            'calculation': calculation
        }
        
        if calculation['will_overflow']:
            result['can_proceed'] = False
            result['warnings'].append({
                'type': 'waste_overflow',
                'message': f'废液桶容量不足！将要废弃 {calculation["total_waste_volume"]:.1f}mL，'
                          f'但废液桶只剩 {calculation["available_capacity"]:.1f}mL 空间，'
                          f'会溢出 {calculation["overflow_volume"]:.1f}mL',
                'severity': 'high',
                'suggestion': '请先清空废液桶或减少废弃的试管数量'
            })
        elif calculation['available_capacity'] < 50:  # 剩余容量少于50mL时预警
            result['warnings'].append({
                'type': 'waste_capacity_warning', 
                'message': f'废液桶即将满载！废弃后仅剩 {calculation["available_capacity"] - calculation["total_waste_volume"]:.1f}mL 空间',
                'severity': 'medium',
                'suggestion': '建议尽快清空废液桶'
            })
        
        return result
    
    def dispose_tubes(self, tubes_to_waste: list, force_disposal: bool = False) -> dict:
        """
        执行试管废弃操作
        
        :param tubes_to_waste: 要废弃的试管列表
        :param force_disposal: 是否强制废弃（忽略容量警告）
        :return: 废弃结果
        """
        # 检查容量
        capacity_check = self.check_waste_capacity_before_disposal(tubes_to_waste)
        
        if not capacity_check['can_proceed'] and not force_disposal:
            return {
                'success': False,
                'reason': 'insufficient_capacity',
                'warnings': capacity_check['warnings'],
                'message': '废液桶容量不足，废弃操作已取消'
            }
        
        # 执行废弃
        total_volume = capacity_check['calculation']['total_waste_volume']
        
        # 记录废弃操作
        disposal_record = {
            'timestamp': time.time(),
            'tubes': tubes_to_waste.copy(),
            'total_volume': total_volume,
            'force_disposal': force_disposal
        }
        self.waste_log.append(disposal_record)
        
        # 更新废液体积
        self.current_waste_volume += total_volume
        
        # 检查是否超出容积（强制废弃的情况）
        if self.current_waste_volume > self.waste_container_capacity:
            overflow = self.current_waste_volume - self.waste_container_capacity
            alarm_data = {
                'alarm_type': 'waste_overflow',
                'message': f'废液桶已溢出 {overflow:.1f}mL！请立即处理',
                'severity': 'critical'
            }
            send_frontend_alarm(alarm_data)
        
        return {
            'success': True,
            'disposed_volume': total_volume,
            'disposed_tubes': len(tubes_to_waste),
            'current_waste_volume': self.current_waste_volume,
            'remaining_capacity': max(0, self.waste_container_capacity - self.current_waste_volume),
            'warnings': capacity_check.get('warnings', [])
        }
    
    def handle_liquid_level_alarm(self) -> dict:
        """
        处理液位检测器报警（废液桶满载）
        
        :return: 处理结果
        """
        # 标记废液桶已满
        self.current_waste_volume = self.waste_container_capacity
        
        alarm_data = {
            'alarm_type': 'waste_container_full',
            'message': '液位检测器报警：废液桶已满！实验已暂停',
            'severity': 'critical',
            'action_required': '请立即清空废液桶后重新启动实验',
            'experiment_action': 'pause'
        }
        
        # 发送报警
        send_frontend_alarm(alarm_data)
        
        return alarm_data
    
    def get_waste_status(self) -> dict:
        """获取废液状态"""
        usage_percent = (self.current_waste_volume / self.waste_container_capacity) * 100 if self.waste_container_capacity > 0 else 0
        
        return {
            'capacity': self.waste_container_capacity,
            'current_volume': self.current_waste_volume,
            'remaining_capacity': max(0, self.waste_container_capacity - self.current_waste_volume),
            'usage_percent': usage_percent,
            'is_full': self.current_waste_volume >= self.waste_container_capacity,
            'is_nearly_full': usage_percent > 90,
            'disposal_count': len(self.waste_log)
        }
```

### 3.5 异常处理与恢复机制

#### 3.5.1 分级异常处理
基于系统安全性和实验连续性的考虑，异常处理分为四个级别：

##### 异常级别定义：
- **提示级(Info)**：不影响实验进行的信息提醒
- **警告级(Warning)**：需要注意但可继续运行的状况
- **错误级(Error)**：无法通过用户主动修复，需要立即危险处理的情况
- **严重级(Critical)**：用户可以主动修复但需要人工干预的问题

```python
class ExceptionHandler:
    
    # 异常级别常量
    LEVEL_INFO = 1      # 提示级
    LEVEL_WARNING = 2   # 警告级  
    LEVEL_ERROR = 3     # 错误级
    LEVEL_CRITICAL = 4  # 严重级
    
    def __init__(self, redis_client, liquid_manager, tube_manager):
        self.redis_client = redis_client
        self.liquid_manager = liquid_manager
        self.tube_manager = tube_manager
        self.experiment_paused = False
        self.emergency_stopped = False
        
    def handle_exception(self, exception_type: str, severity: int, details: dict = None):
        """
        分级异常处理核心方法
        
        :param exception_type: 异常类型
        :param severity: 严重程度 (1-4)
        :param details: 异常详细信息
        """
        exception_data = {
            'type': exception_type,
            'severity': severity,
            'timestamp': time.time(),
            'details': details or {}
        }
        
        if severity == self.LEVEL_INFO:  # 提示级
            self._handle_info_level(exception_data)
            
        elif severity == self.LEVEL_WARNING:  # 警告级
            self._handle_warning_level(exception_data)
            
        elif severity == self.LEVEL_ERROR:  # 错误级
            self._handle_error_level(exception_data)
            
        elif severity == self.LEVEL_CRITICAL:  # 严重级
            self._handle_critical_level(exception_data)
    
    def _handle_info_level(self, exception_data):
        """
        提示级异常处理
        - 仅记录日志和通知用户
        - 实验继续正常进行
        """
        self._log_exception(exception_data, "INFO")
        self._notify_frontend({
            'level': 'info',
            'message': self._format_message(exception_data),
            'dismissible': True,
            'auto_close': 5000  # 5秒后自动关闭
        })
    
    def _handle_warning_level(self, exception_data):
        """
        警告级异常处理
        - 记录日志并发出警告
        - 尝试自动调整参数
        - 实验继续进行但需监控
        """
        self._log_exception(exception_data, "WARNING")
        
        # 尝试自动调整
        adjustment_made = self._try_auto_adjustment(exception_data)
        
        # 通知前端
        self._notify_frontend({
            'level': 'warning',
            'message': self._format_message(exception_data),
            'adjustment': adjustment_made,
            'dismissible': True,
            'persistent': True  # 持续显示直到用户确认
        })
    
    def _handle_error_level(self, exception_data):
        """
        错误级异常处理
        - 暂停实验
        - 保存当前状态
        - 等待用户处理
        """
        if not self.experiment_paused:
            self._pause_experiment()
            self._save_current_state()
        
        self._log_exception(exception_data, "ERROR")
        
        # 通知前端并要求用户操作
        self._notify_frontend({
            'level': 'error',
            'message': self._format_message(exception_data),
            'actions_required': self._get_required_actions(exception_data),
            'dismissible': False,
            'modal': True  # 以模态框形式显示
        })
        
        # 等待用户响应
        self._wait_for_user_resolution(exception_data)
    
    def _handle_critical_level(self, exception_data):
        """
        严重级异常处理
        - 立即紧急停止
        - 保存所有数据
        - 安全关闭系统
        """
        if not self.emergency_stopped:
            self._emergency_stop()
            self._save_all_data()
        
        self._log_exception(exception_data, "CRITICAL")
        
        # 发送紧急通知
        self._notify_frontend({
            'level': 'critical',
            'message': self._format_message(exception_data),
            'emergency': True,
            'dismissible': False,
            'sound_alarm': True
        })
        
        # 通知管理员
        self._notify_admin(exception_data)
        
        # 启动安全关闭程序
        self._safe_shutdown()

class SmartExceptionHandler(ExceptionHandler):
    """智能异常处理器，根据异常类型自动判断处理策略"""
    
    def __init__(self, redis_client, liquid_manager, tube_manager):
        super().__init__(redis_client, liquid_manager, tube_manager)
        self.exception_definitions = EXCEPTION_DEFINITIONS
        
    def handle_smart_exception(self, exception_type: str, context: dict = None):
        """
        智能异常处理入口
        
        :param exception_type: 异常类型（在EXCEPTION_DEFINITIONS中定义）
        :param context: 异常上下文信息
        """
        if exception_type not in self.exception_definitions:
            # 未定义的异常类型，默认为警告级
            self.handle_exception(exception_type, self.LEVEL_WARNING, context)
            return
        
        exception_def = self.exception_definitions[exception_type]
        severity = exception_def['level']
        
        # 补充异常详细信息
        details = {
            'description': exception_def['description'],
            'context': context or {}
        }
        
        if 'auto_adjustment' in exception_def:
            details['auto_adjustment'] = exception_def['auto_adjustment']
        if 'required_actions' in exception_def:
            details['required_actions'] = exception_def['required_actions']
        if 'emergency_action' in exception_def:
            details['emergency_action'] = exception_def['emergency_action']
            
        self.handle_exception(exception_type, severity, details)
    
    def _try_auto_adjustment(self, exception_data):
        """尝试自动调整参数"""
        exception_type = exception_data['type']
        
        if exception_type == 'pressure_high':
            # 提醒用户检查压力状况
            self._send_pressure_warning(exception_data['details']['context'])
            return "已提醒用户检查压力状况"
            
        elif exception_type == 'solvent_low_volume':
            # 发送补液提醒
            self._send_refill_reminder(exception_data['details']['context'])
            return "已发送补液提醒"
            
        elif exception_type == 'flow_rate_deviation':
            # 提醒用户检查泵系统
            self._send_pump_check_reminder(exception_data['details']['context'])
            return "已提醒用户检查泵系统"
        
        return None
    
    # 使用示例
    def example_usage(self):
        """异常处理使用示例"""
        
        # 检测到峰 - 提示级
        self.handle_smart_exception('peak_detected', {
            'peak_id': 3,
            'retention_time': 15.2,
            'tube_number': 5
        })
        
        # 原液不足 - 警告级
        self.handle_smart_exception('solvent_low_volume', {
            'solvent': 'A',
            'remaining_volume': 2.5,
            'estimated_time_left': 8.5
        })
        
        # 压力超限 - 错误级

            'current_pressure': 250,
            'max_pressure': 200,
            'column_model': 'FC003025'
        })
        
        # 原液耗尽 - 严重级

            'solvent': 'A',
            'detection_time': time.time()
        })
```

#### 4.2.5 试管-检测器数据关联查询

**根据UV检测时间点找到对应的试管：**
```sql
-- 查找特定时间点UV数据对应的试管
WITH tube_timeline AS (
    SELECT 
        experiment_id,
        json_array_elements(tube_switching_timeline) as tube_info
    FROM experiments 
    WHERE id = 'experiment_uuid'
),
expanded_timeline AS (
    SELECT 
        experiment_id,
        (tube_info->>'tube_id') as tube_id,
        (tube_info->>'tube_number')::integer as tube_number,
        (tube_info->>'switch_in')::timestamp as switch_in_time,
        (tube_info->>'switch_out')::timestamp as switch_out_time
    FROM tube_timeline
)
SELECT 
    rd.timestamp,
    rd.uv_absorption,
    et.tube_id,
    et.tube_number
FROM realtime_data rd
JOIN expanded_timeline et ON rd.experiment_id = et.experiment_id
WHERE rd.timestamp >= et.switch_in_time 
  AND rd.timestamp < et.switch_out_time
  AND rd.timestamp BETWEEN '2024-01-01 10:15:00' AND '2024-01-01 10:15:10'
ORDER BY rd.timestamp;
```

**分析每个试管收集的UV曲线片段：**
```sql
-- 获取每个试管的完整UV曲线
SELECT 
    f.tube_id,
    f.tube_number,
    array_agg(rd.uv_absorption ORDER BY rd.timestamp) as uv_curve,
    array_agg(rd.timestamp ORDER BY rd.timestamp) as time_points,
    f.switch_in_time,
    f.switch_out_time,
    extract(epoch from (f.switch_out_time - f.switch_in_time)) as collection_duration_seconds
FROM fractions f
JOIN realtime_data rd ON f.experiment_id = rd.experiment_id
WHERE rd.timestamp >= f.switch_in_time 
  AND rd.timestamp < f.switch_out_time
  AND f.experiment_id = 'experiment_uuid'
GROUP BY f.tube_id, f.tube_number, f.switch_in_time, f.switch_out_time
ORDER BY f.tube_number;
```

**检测数据完整性：**
```sql
-- 检查每个试管收集期间的数据完整性
SELECT 
    f.tube_id,
    f.tube_number,
    count(rd.id) as actual_data_points,
    extract(epoch from (f.switch_out_time - f.switch_in_time)) as expected_seconds,
    CASE 
        WHEN count(rd.id) >= extract(epoch from (f.switch_out_time - f.switch_in_time)) * 0.95 
        THEN 'complete'
        ELSE 'incomplete'
    END as data_completeness
FROM fractions f
LEFT JOIN realtime_data rd ON f.experiment_id = rd.experiment_id
    AND rd.timestamp >= f.switch_in_time 
    AND rd.timestamp < f.switch_out_time
WHERE f.experiment_id = 'experiment_uuid'
GROUP BY f.tube_id, f.tube_number, f.switch_in_time, f.switch_out_time
ORDER BY f.tube_number;
```

**JavaScript实现：**
```javascript
class ChromatogramChart {
  constructor(containerId, data) {
    this.container = d3.select(containerId);
    this.data = data;
    this.margin = {top: 20, right: 80, bottom: 60, left: 80};
    this.width = 1000 - this.margin.left - this.margin.right;
    this.height = 500 - this.margin.top - this.margin.bottom;
    
    this.initChart();
    this.setupScales();
    this.drawChart();
    this.setupInteractions();
  }

  initChart() {
    // 清空容器
    this.container.selectAll("*").remove();
    
    // 创建SVG
    this.svg = this.container
      .append("svg")
      .attr("width", this.width + this.margin.left + this.margin.right)
      .attr("height", this.height + this.margin.top + this.margin.bottom);
      
    // 创建主绘图区域
    this.g = this.svg.append("g")
      .attr("transform", `translate(${this.margin.left},${this.margin.top})`);
      
    // 创建剪切路径
    this.g.append("defs")
      .append("clipPath")
      .attr("id", "clip")
      .append("rect")
      .attr("width", this.width)
      .attr("height", this.height);

    // 创建tooltip
    this.tooltip = d3.select("body").append("div")
      .attr("class", "tooltip")
      .style("opacity", 0);
  }

  setupScales() {
    // 时间比例尺（X轴）
    this.xScale = d3.scaleLinear()
      .domain([0, this.data.total_duration])
      .range([0, this.width]);

    // UV值比例尺（左Y轴）
    const uvMax = d3.max(this.data.uv_data, d => Math.max(d.uv_280nm, d.uv_254nm));
    this.yScaleUV = d3.scaleLinear()
      .domain([0, uvMax * 1.1])
      .range([this.height, 0]);

    // 梯度比例尺（右Y轴）
    this.yScaleGradient = d3.scaleLinear()
      .domain([0, 100])
      .range([this.height, 0]);

    // 缩放行为
    this.zoom = d3.zoom()
      .scaleExtent([0.5, 50])
      .extent([[0, 0], [this.width, this.height]])
      .on("zoom", (event) => this.zoomed(event));

    this.svg.call(this.zoom);
  }

  drawChart() {
    this.drawGrid();
    this.drawAxes();
    this.drawTubeSegments();
    this.drawUVCurve();
    this.drawGradientCurve();
    this.drawPeaks();
    this.drawLegend();
  }

  drawGrid() {
    // X轴网格线
    this.g.selectAll(".grid-line-x")
      .data(this.xScale.ticks(20))
      .enter()
      .append("line")
      .attr("class", "grid-line")
      .attr("x1", d => this.xScale(d))
      .attr("x2", d => this.xScale(d))
      .attr("y1", 0)
      .attr("y2", this.height);

    // Y轴网格线
    this.g.selectAll(".grid-line-y")
      .data(this.yScaleUV.ticks(10))
      .enter()
      .append("line")
      .attr("class", "grid-line")
      .attr("x1", 0)
      .attr("x2", this.width)
      .attr("y1", d => this.yScaleUV(d))
      .attr("y2", d => this.yScaleUV(d));
  }

  drawAxes() {
    // X轴
    this.g.append("g")
      .attr("class", "axis")
      .attr("transform", `translate(0,${this.height})`)
```

```javascript
class PreExperimentConfigManager {
  constructor() {
    this.wasteContainers = [];
    this.selectedWasteContainer = null;
    this.experimentMode = 'normal';
    
    this.initializeInterface();
    this.loadWasteContainers();
  }

  initializeInterface() {
    // 监听实验模式切换
    document.querySelectorAll('input[name="experiment_mode"]').forEach(radio => {
      radio.addEventListener('change', (e) => {
        this.experimentMode = e.target.value;
        this.toggleModeConfiguration();
      });
    });
    
    // 监听试管范围变化
    document.getElementById('start-tube').addEventListener('change', () => {
      this.updateEstimatedVolume();
    });
    
    document.getElementById('end-tube').addEventListener('change', () => {
      this.updateEstimatedVolume();
    });
  }

  async loadWasteContainers() {
    try {
      const response = await fetch('/api/waste-containers');
      const data = await response.json();
      this.wasteContainers = data.waste_containers;
      this.renderWasteContainers();
    } catch (error) {
      console.error('加载废液桶列表失败:', error);
    }
  }

  renderWasteContainers() {
    const grid = document.querySelector('.waste-container-grid');
    grid.innerHTML = '';
    
    this.wasteContainers.forEach(container => {
      const containerElement = document.createElement('div');
      containerElement.className = 'waste-container-item';
      containerElement.innerHTML = `
        <div class="container-info">
          <h4>${container.container_name}</h4>
          <div class="container-details">
            <span class="container-type">${container.container_type}</span>
            <span class="container-location">${container.location}</span>
          </div>
          <div class="capacity-info">
            <div class="capacity-bar-small">
              <div class="capacity-fill-small" 
                   style="width: ${container.fill_percentage}%"></div>
            </div>
            <span class="capacity-text">
              ${container.current_volume_ml}/${container.max_volume_ml} ml
            </span>
          </div>
        </div>
        <button class="select-container-btn" 
                onclick="preExpConfig.selectWasteContainer('${container.id}')"
                ${container.status !== 'active' ? 'disabled' : ''}>
          选择
        </button>
      `;
      
      if (container.status !== 'active') {
        containerElement.classList.add('disabled');
      }
      
      grid.appendChild(containerElement);
    });
  }

  toggleModeConfiguration() {
    const wasteConfig = document.querySelector('.waste-mode-config');
    const tubeConfig = document.querySelector('.tube-mode-config');
    
    if (this.experimentMode === 'waste') {
      wasteConfig.style.display = 'block';
      tubeConfig.style.display = 'none';
    } else {
      wasteConfig.style.display = 'none';
      tubeConfig.style.display = 'block';
    }
  }

  selectWasteContainer(containerId) {
    const container = this.wasteContainers.find(c => c.id === containerId);
    if (!container) return;
    
    this.selectedWasteContainer = container;
    
    // 更新选择状态
    document.querySelectorAll('.waste-container-item').forEach(item => {
      item.classList.remove('selected');
    });
    
    event.target.closest('.waste-container-item').classList.add('selected');
    
    // 更新废液桶信息显示
    this.updateWasteContainerInfo();
    this.updateEstimatedVolume();
  }

  updateWasteContainerInfo() {
    if (!this.selectedWasteContainer) return;
    
    const container = this.selectedWasteContainer;
    const remainingCapacity = container.max_volume_ml - container.current_volume_ml;
    
    document.getElementById('remaining-capacity').textContent = remainingCapacity;
    
    const fillBar = document.getElementById('capacity-fill-bar');
    fillBar.style.width = container.fill_percentage + '%';
  }

  updateEstimatedVolume() {
    if (this.experimentMode === 'normal') {
      // 正常模式：根据试管数量估算
      const startTube = parseInt(document.getElementById('start-tube').value);
      const endTube = parseInt(document.getElementById('end-tube').value);
      const tubeCount = Math.max(0, endTube - startTube + 1);
      const estimatedVolume = tubeCount * 10; // 假设每管10ml
      
      document.getElementById('estimated-waste-volume').textContent = estimatedVolume;
    } else {
      // 废弃模式：所有液体都流入废液桶
      const estimatedVolume = 300; // 假设整个实验300ml
      document.getElementById('estimated-waste-volume').textContent = estimatedVolume;
    }
  }

  async confirm() {
    const configData = {
      experiment_name: document.getElementById('experiment-name').value || '未命名实验',
      waste_mode: this.experimentMode === 'waste',
      enable_column_conditioning: document.getElementById('enable-conditioning').checked,
      enable_sample_injection: document.getElementById('enable-injection').checked
    };

    if (this.experimentMode === 'waste') {
      if (!this.selectedWasteContainer) {
        alert('请选择废液桶');
        return;
      }
      configData.waste_container_id = this.selectedWasteContainer.id;
    } else {
      configData.start_tube_number = parseInt(document.getElementById('start-tube').value);
      configData.end_tube_number = parseInt(document.getElementById('end-tube').value);
    }

    try {
      const response = await fetch('/api/pre-experiment-configs', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify(configData)
      });

      const result = await response.json();
      if (result.success) {
        alert('实验配置创建成功！');
        // 跳转到实验界面或其他操作
      } else {
        alert('配置创建失败: ' + result.message);
      }
    } catch (error) {
      alert('配置创建失败: ' + error.message);
    }
  }

  cancel() {
    // 取消配置，返回主界面
    window.history.back();
  }
}
```

```http
GET /api/experiments/{experiment_id}/gradient/status
```

```json
{
  "method_type": "gradient",
  "current_time_seconds": 1245.6,
  "current_gradient": {
    "solvent_a_percent": 65.3,
    "solvent_b_percent": 34.7
  },
  "modifiable": true,
  "next_gradient_point": {
    "time_seconds": 1800,
    "solvent_b_percent": 80.0
  }
}
```

**响应示例（自动梯度）：**
```json
{
  "method_type": "auto_gradient",
  "current_time_seconds": 892.3,
  "current_gradient": {
    "solvent_a_percent": 45.2,
    "solvent_b_percent": 54.8
  },
  "gradient_rate": 2.5,
  "manual_hold": false,
  "modifiable": true
}
```

```http
POST /api/experiments/{experiment_id}/gradient/modify
```

**JavaScript实现：**
```javascript
class ExperimentController {
  constructor(experimentId) {
    this.experimentId = experimentId;
    this.currentState = 'unknown';
    this.gradientType = 'unknown';
    this.setupEventListeners();
    this.startStatusPolling();
  }

  setupEventListeners() {
    // 实验控制按钮
    document.getElementById('pause-btn').addEventListener('click', () => {
      this.pauseExperiment();
    });
    
    document.getElementById('resume-btn').addEventListener('click', () => {
      this.resumeExperiment();
    });
    
    document.getElementById('terminate-btn').addEventListener('click', () => {
      if (confirm('确定要终止当前实验吗？此操作不可撤销。')) {
        this.terminateExperiment();
      }
    });

    // 标准梯度控制
    document.getElementById('gradient-slider').addEventListener('input', (e) => {
      document.getElementById('gradient-input').value = e.target.value;
    });
    
    document.getElementById('gradient-input').addEventListener('input', (e) => {
      document.getElementById('gradient-slider').value = e.target.value;
    });
    
    document.getElementById('apply-gradient').addEventListener('click', () => {
      this.modifyGradient();
    });

    // 自动梯度控制
    document.getElementById('apply-auto-gradient').addEventListener('click', () => {
      this.modifyAutoGradient();
    });
  }

  startStatusPolling() {
    // 每秒更新一次状态
    setInterval(() => {
      this.updateGradientStatus();
    }, 1000);
  }

  async updateGradientStatus() {
    try {
      const response = await fetch(`/api/experiments/${this.experimentId}/gradient/status`);
      const data = await response.json();
      
      this.updateStatusDisplay(data);
      this.updateGradientControls(data);
      
    } catch (error) {
      console.error('获取梯度状态失败:', error);
    }
  }

  updateStatusDisplay(data) {
    // 更新时间显示
    const minutes = Math.floor(data.current_time_seconds / 60);
    const seconds = Math.floor(data.current_time_seconds % 60);
    document.getElementById('current-time').textContent = 
      `${minutes}:${seconds.toString().padStart(2, '0')}`;
    
    // 更新比例显示
    if (data.current_gradient) {
      document.getElementById('current-a').textContent = 
        data.current_gradient.solvent_a_percent.toFixed(1);
      document.getElementById('current-b').textContent = 
        data.current_gradient.solvent_b_percent.toFixed(1);
    }
  }

  updateGradientControls(data) {
    // 根据方法类型显示相应控制面板
    if (data.method_type === 'gradient') {
      this.showStandardGradientControls(data);
      this.gradientType = 'standard';
    } else if (data.method_type === 'auto_gradient') {
      this.showAutoGradientControls(data);
      this.gradientType = 'auto';
    } else {
      this.hideGradientControls();
    }
  }

  showStandardGradientControls(data) {
    document.getElementById('standard-gradient-controls').style.display = 'block';
    document.getElementById('auto-gradient-controls').style.display = 'none';
    
    const currentB = data.current_gradient.solvent_b_percent;
    document.getElementById('gradient-slider').value = currentB;
    document.getElementById('gradient-input').value = currentB.toFixed(1);
  }

  showAutoGradientControls(data) {
    document.getElementById('standard-gradient-controls').style.display = 'none';
    document.getElementById('auto-gradient-controls').style.display = 'block';
    
    document.getElementById('auto-gradient-input').value = 
      data.current_gradient.solvent_b_percent.toFixed(1);
    document.getElementById('gradient-rate-input').value = data.gradient_rate;
    document.getElementById('manual-hold-checkbox').checked = data.manual_hold;
  }

  hideGradientControls() {
    document.getElementById('standard-gradient-controls').style.display = 'none';
    document.getElementById('auto-gradient-controls').style.display = 'none';
  }

  async pauseExperiment() {
    try {
      const response = await fetch(`/api/experiments/${this.experimentId}/control`, {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({action: 'pause'})
      });
      
      const result = await response.json();
      if (result.success) {
        this.updateControlButtons('paused');
        this.showNotification(result.message, 'success');
      } else {
        this.showNotification(result.error || '暂停失败', 'error');
      }
    } catch (error) {
      this.showNotification('暂停请求失败', 'error');
    }
  }

  async resumeExperiment() {
    try {
      const response = await fetch(`/api/experiments/${this.experimentId}/control`, {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({action: 'resume'})
      });
      
      const result = await response.json();
      if (result.success) {
        this.updateControlButtons('running');
        this.showNotification(result.message, 'success');
      } else {
        this.showNotification(result.error || '继续失败', 'error');
      }
    } catch (error) {
      this.showNotification('继续请求失败', 'error');
    }
  }

  async terminateExperiment() {
    try {
      const response = await fetch(`/api/experiments/${this.experimentId}/control`, {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({action: 'terminate'})
      });
      
      const result = await response.json();
      if (result.success) {
        this.updateControlButtons('terminated');
        this.showNotification(result.message, 'info');
      } else {
        this.showNotification(result.error || '终止失败', 'error');
      }
    } catch (error) {
      this.showNotification('终止请求失败', 'error');
    }
  }

  async modifyGradient() {
    const newB = parseFloat(document.getElementById('gradient-input').value);
    
    if (isNaN(newB) || newB < 0 || newB > 100) {
      this.showNotification('请输入有效的B比例 (0-100)', 'error');
      return;
    }

    try {
      const response = await fetch(`/api/experiments/${this.experimentId}/gradient/modify`, {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({
          new_solvent_b_percent: newB,
          apply_immediately: true
        })
      });
      
      const result = await response.json();
      if (result.success) {
        this.showNotification(result.message, 'success');
      } else {
        this.showNotification(result.error || '修改失败', 'error');
      }
    } catch (error) {
      this.showNotification('修改请求失败', 'error');
    }
  }

  async modifyAutoGradient() {
    const newB = parseFloat(document.getElementById('auto-gradient-input').value);
    const newRate = parseFloat(document.getElementById('gradient-rate-input').value);
    const manualHold = document.getElementById('manual-hold-checkbox').checked;

    const modifications = {};
    
    if (!isNaN(newB) && newB >= 0 && newB <= 100) {
      modifications.new_solvent_b_percent = newB;
    }
    
    if (!isNaN(newRate) && newRate >= 0.1 && newRate <= 10) {
      modifications.new_gradient_rate = newRate;
    }
    
    modifications.manual_hold = manualHold;

    try {
      const response = await fetch(`/api/experiments/${this.experimentId}/auto-gradient/modify`, {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify(modifications)
      });
      
      const result = await response.json();
      if (result.success) {
        this.showNotification(result.message, 'success');
      } else {
        this.showNotification(result.error || '修改失败', 'error');
      }
    } catch (error) {
      this.showNotification('修改请求失败', 'error');
    }
  }

  updateControlButtons(state) {
    const pauseBtn = document.getElementById('pause-btn');
    const resumeBtn = document.getElementById('resume-btn');
    const terminateBtn = document.getElementById('terminate-btn');
    
    switch (state) {
      case 'running':
        pauseBtn.disabled = false;
        resumeBtn.disabled = true;
        terminateBtn.disabled = false;
        break;
      case 'paused':
        pauseBtn.disabled = true;
        resumeBtn.disabled = false;
        terminateBtn.disabled = false;
        break;
      case 'terminated':
        pauseBtn.disabled = true;
        resumeBtn.disabled = true;
        terminateBtn.disabled = true;
        break;
    }
    
    this.currentState = state;
    document.getElementById('experiment-status').textContent = 
      state === 'running' ? '运行中' : 
      state === 'paused' ? '已暂停' : '已终止';
  }

  showNotification(message, type) {
    // 简单的通知显示，可以集成更复杂的通知系统
    const notification = document.createElement('div');
    notification.className = `notification ${type}`;
    notification.textContent = message;
    
    document.body.appendChild(notification);
    
    setTimeout(() => {
      notification.remove();
    }, 5000);
  }
}

// 初始化实验控制器
const experimentController = new ExperimentController(window.experimentId);
```

**JavaScript实现：**
```javascript
class PostExperimentManager {
  constructor(experimentId) {
    this.experimentId = experimentId;
    this.selectedTubes = new Set();
    this.tasks = [];
    this.experimentData = null;
    
    this.init();
  }

  async init() {
    await this.loadExperimentSummary();
    this.setupEventListeners();
    this.renderInterface();
  }

  async loadExperimentSummary() {
    try {
      const response = await fetch(`/api/experiments/${this.experimentId}/summary`);
      this.experimentData = await response.json();
    } catch (error) {
      console.error('加载实验数据失败:', error);
    }
  }

  setupEventListeners() {
    // 试管选择事件
    document.addEventListener('click', (e) => {
      if (e.target.classList.contains('tube-button')) {
        this.toggleTubeSelection(parseInt(e.target.dataset.tubeNumber));
      }
    });

    // 峰点击事件
    document.addEventListener('click', (e) => {
      if (e.target.classList.contains('peak-item')) {
        this.highlightPeakTubes(e.target.dataset.peakId);
      }
    });

    // 任务创建按钮
    document.getElementById('create-keep-task').addEventListener('click', () => {
      this.createTask('keep');
    });

    document.getElementById('create-discard-task').addEventListener('click', () => {
      this.createTask('discard');
    });

    // 批量执行按钮
    document.getElementById('execute-all-tasks').addEventListener('click', () => {
      this.executeAllTasks();
    });

    // 系统操作按钮
    document.getElementById('system-reset-btn').addEventListener('click', () => {
      this.systemReset();
    });

    document.getElementById('cleaning-setup-btn').addEventListener('click', () => {
      this.showCleaningModal();
    });

    // 清洗相关按钮
    document.getElementById('start-cleaning').addEventListener('click', () => {
      this.startCleaning();
    });

    document.getElementById('cancel-cleaning').addEventListener('click', () => {
      this.hideCleaningModal();
    });
  }

  renderInterface() {
    this.renderSummary();
    this.renderTubeGrid();
    this.renderPeakList();
    this.renderCleaningTubeGrid();
  }

  renderSummary() {
    const exp = this.experimentData.experiment;
    document.getElementById('total-peaks').textContent = exp.total_peaks;
    document.getElementById('total-tubes').textContent = exp.total_fractions;

  async createTask(operationType) {
    if (this.selectedTubes.size === 0) {
      alert('请先选择试管');
      return;
    }

    const selectedTubesArray = Array.from(this.selectedTubes);
    
    try {
      const response = await fetch(`/api/experiments/${this.experimentId}/tasks`, {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({
          task_type: 'tube_operation',
          selected_tubes: selectedTubesArray,
          operation_type: operationType,
          peak_references: this.getRelatedPeaks(selectedTubesArray)
        })
      });
      
      const result = await response.json();
      if (result.success) {
        this.tasks.push({
          id: result.task_id,
          tubes: selectedTubesArray,
          operation: operationType,
          status: 'pending'
        });
        
        this.renderTaskList();
        this.selectedTubes.clear();
        this.updateTubeDisplays();
        this.updateSelectedTubesDisplay();
        
        this.showNotification(`${operationType === 'keep' ? '保留' : '废弃'}任务已创建`, 'success');
      }
    } catch (error) {
      this.showNotification('创建任务失败', 'error');
    }
  }

    Object.entries(this.experimentData.peak_tube_mapping).forEach(([peakId, tubes]) => {
      if (tubes.some(tube => tubeNumbers.includes(tube))) {
        relatedPeaks.push(peakId);

  async executeAllTasks() {
    if (this.tasks.length === 0) {
      alert('没有待执行的任务');
      return;
    }

    const pendingTasks = this.tasks.filter(t => t.status === 'pending');
    if (pendingTasks.length === 0) {
      alert('没有待执行的任务');
      return;
    }

    try {
      const taskIds = pendingTasks.map(t => t.id);
      
      const response = await fetch(`/api/experiments/${this.experimentId}/batch-tasks`, {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({
          task_ids: taskIds,
          batch_name: `批量处理 ${taskIds.length} 个任务`,
          execution_mode: 'sequential',
          continue_on_error: true
        })
      });
      
      const batchResult = await response.json();
      if (batchResult.success) {
        // 执行批处理任务
        const execResponse = await fetch(`/api/batch-tasks/${batchResult.batch_id}/execute`, {
          method: 'POST'
        });
        
        const execResult = await execResponse.json();
        this.handleBatchExecutionResult(execResult);
      }
    } catch (error) {
      this.showNotification('批量执行失败', 'error');
    }
  }

        method: 'POST'

        this.showNotification('系统重置完成', 'success');

        this.showNotification('系统重置失败', 'error');

    } catch (error) {

    }
  }
```

```python
        # 3. 推送到前端
        self.notify_frontend(alarm_data)

        self.send_notification(alarm_data)
```

```python
class PreExperimentController:
    def __init__(self, db_connection, tube_manager, method_manager):
        self.db = db_connection
        self.tube_manager = tube_manager
        self.method_manager = method_manager
    
    def create_experiment_config(self, user_id, config_data):
        """
        创建实验前配置
        
        config_data包含:
        - experiment_name: 实验名称
        - start_tube_number: 开始试管号
        - end_tube_number: 结束试管号
        - selected_methods: 选择的方法列表
        - method_tube_assignments: 方法-试管分配
        - enable_column_conditioning: 是否润柱
        - enable_sample_injection: 是否进样
        - waste_mode: 是否为废弃模式
        - waste_container_id: 废液桶ID(废弃模式时必需)
        """
        
        # 1. 验证废弃模式配置
        if config_data.get('waste_mode', False):
            if not config_data.get('waste_container_id'):
                raise ValueError("废弃模式下必须指定废液桶ID")
            
            # 验证废液桶状态
            waste_container = self.db.fetchone("""
                SELECT status, current_volume_ml, max_volume_ml, container_type
                FROM waste_containers WHERE id = %s
            """, (config_data['waste_container_id'],))
            
            if not waste_container:
                raise ValueError("指定的废液桶不存在")
            
            if waste_container['status'] != 'active':
                raise ValueError(f"废液桶状态异常: {waste_container['status']}")
            
            if waste_container['current_volume_ml'] >= waste_container['max_volume_ml'] * 0.9:
                raise ValueError("废液桶接近满容量，请先清空")
        
        # 2. 验证试管范围和可用性（废弃模式下跳过试管检查）
        if not config_data.get('waste_mode', False):
            available_tubes = self.tube_manager.get_available_tubes()
            requested_range = range(config_data['start_tube_number'], 
                                  config_data['end_tube_number'] + 1)
            
            unavailable_tubes = [t for t in requested_range if t not in available_tubes]
            if unavailable_tubes:
                raise ValueError(f"试管 {unavailable_tubes} 不可用，请检查状态")
        else:
            # 废弃模式下设置虚拟试管范围
            requested_range = range(1, 2)  # 不使用实际试管
        
        # 2. 验证方法有效性
        for method_id in config_data['selected_methods']:
            if not self.method_manager.is_method_valid(method_id):
                raise ValueError(f"方法 {method_id} 无效或已停用")
        
        # 3. 检查清洗需求
        cleaning_required_tubes = self.tube_manager.get_tubes_requiring_cleaning(
            list(requested_range)
        )
        
        if cleaning_required_tubes:
            config_data['pre_cleaning_required'] = True
            config_data['selected_cleaning_tubes'] = cleaning_required_tubes
        
        # 4. 自动分配方法到试管（如果未指定）
        if not config_data.get('method_tube_assignments'):
            config_data['method_tube_assignments'] = self._auto_assign_methods_to_tubes(
                config_data['selected_methods'], 
                list(requested_range)
            )
        
        # 5. 保存配置到数据库
        config_id = self.db.execute("""
            INSERT INTO pre_experiment_configs 
            (experiment_name, operator_id, start_tube_number, end_tube_number,
             selected_tubes, selected_methods, method_tube_assignments,
             enable_column_conditioning, enable_sample_injection,
             waste_mode, waste_container_id,
             pre_cleaning_required, selected_cleaning_tubes)
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
            RETURNING id
        """, (
            config_data['experiment_name'], user_id,
            config_data['start_tube_number'], config_data['end_tube_number'],
            list(requested_range), config_data['selected_methods'],
            json.dumps(config_data['method_tube_assignments']),
            config_data.get('enable_column_conditioning', True),
            config_data.get('enable_sample_injection', True),
            config_data.get('waste_mode', False),
            config_data.get('waste_container_id'),
            config_data.get('pre_cleaning_required', False),
            config_data.get('selected_cleaning_tubes', [])
        )).fetchone()[0]
        
        return config_id
    
    def _auto_assign_methods_to_tubes(self, methods, tubes):
        """自动分配方法到试管 - 默认所有试管使用所有方法"""
        assignments = {}
        for method_id in methods:
            assignments[method_id] = tubes
        return assignments
    
    def validate_experiment_readiness(self, config_id):
        """验证实验准备就绪状态"""
        config = self.db.execute("""
            SELECT * FROM pre_experiment_configs WHERE id = %s
        """, (config_id,)).fetchone()
        
        readiness_check = {
            'ready': True,
            'issues': [],
            'warnings': []
        }
        
        # 1. 检查清洗状态
        if config['pre_cleaning_required']:
            cleaning_status = self._check_cleaning_completion(
                config['selected_cleaning_tubes']
            )
            if not cleaning_status['completed']:
                readiness_check['ready'] = False
                readiness_check['issues'].append(
                    f"试管清洗未完成: {cleaning_status['remaining_tubes']}"
                )
        
        # 2. 检查试管状态
        tube_status = self.tube_manager.get_tubes_status(config['selected_tubes'])
        occupied_tubes = [t for t, s in tube_status.items() if s != 'ready']
        if occupied_tubes:
            readiness_check['ready'] = False
            readiness_check['issues'].append(f"试管占用: {occupied_tubes}")
        
        # 3. 检查溶剂余量
        solvent_status = self._check_solvent_levels(config['selected_methods'])
        if solvent_status['insufficient']:
            readiness_check['warnings'].append(
                f"溶剂余量不足: {solvent_status['low_solvents']}"
            )
        
        return readiness_check
```

#### 10.1.2 实验执行流程控制

```python
class ExperimentExecutionController:
    def __init__(self, hardware_manager, notification_service):
        self.hardware = hardware_manager
        self.notifications = notification_service
        self.current_experiment = None
        self.experiment_state = 'idle'
        
    def start_experiment(self, config_id):
        """启动实验"""
        config = self._load_experiment_config(config_id)
        
        # 更新配置状态为运行中
        self.db.execute("""
            UPDATE pre_experiment_configs 
            SET config_status = 'running', started_at = NOW()
            WHERE id = %s
        """, (config_id,))
        
        self.current_experiment = config
        
        # 根据配置决定执行流程
        if not config['enable_column_conditioning'] and not config['enable_sample_injection']:
            # 直接开始收集
            self._start_direct_collection()
        elif config['enable_column_conditioning'] and not config['enable_sample_injection']:
            # 只润柱，然后直接收集
            self._start_column_conditioning_only()
        else:
            # 标准流程：润柱 -> 等待进样 -> 收集
            self._start_standard_flow()
    
    def _start_direct_collection(self):
        """直接开始收集模式"""
        self.experiment_state = 'collecting'
        
        if self.current_experiment.get('waste_mode', False):
            self._notify_user("实验开始", "废弃模式 - 所有液体将流入废液桶")
            
            # 设置阀门切换到废液桶
            waste_container_id = self.current_experiment['waste_container_id']
            self._switch_to_waste_container(waste_container_id)
        else:
            self._notify_user("实验开始", "直接进入收集阶段")
            
            # 初始化收集系统
            self.hardware.collection_module.initialize_collection(
                self.current_experiment['start_tube_number']
            )
        
        # 启动数据采集
        self.hardware.detector.start_data_collection()
        
        # 启动实验主循环
        self._start_experiment_loop()
    
    def _start_column_conditioning_only(self):
        """仅润柱模式"""
        self.experiment_state = 'conditioning'
        self._notify_user("实验开始", "开始润柱，完成后将直接进入收集")
        
        # 执行润柱程序
        conditioning_result = self.hardware.column_manager.condition_column(
            self.current_experiment['selected_methods'][0]  # 使用第一个方法的条件
        )
        
        if conditioning_result['success']:
            self.experiment_state = 'collecting'
            self._start_direct_collection()
        else:
            self._handle_conditioning_failure(conditioning_result)
    
    def _start_standard_flow(self):
        """标准实验流程"""
        if self.current_experiment['enable_column_conditioning']:
            self._execute_column_conditioning()
        else:
            self._wait_for_sample_injection()
    
    def _execute_column_conditioning(self):
        """执行润柱程序"""
        self.experiment_state = 'conditioning'
        
        if self.current_experiment.get('waste_mode', False):
            self._notify_user("润柱开始", "废弃模式 - 润柱液体将流入废液桶")
            
            # 润柱时也切换到废液桶
            waste_container_id = self.current_experiment['waste_container_id']
            self._switch_to_waste_container(waste_container_id)
        else:
            self._notify_user("润柱开始", "正在执行柱平衡程序...")
        
        conditioning_result = self.hardware.column_manager.condition_column(
            self.current_experiment['selected_methods'][0],
            waste_mode=self.current_experiment.get('waste_mode', False)
        )
        
        if conditioning_result['success']:
            if self.current_experiment['enable_sample_injection']:
                self._wait_for_sample_injection()
            else:
                self._start_direct_collection()
        else:
            self._handle_conditioning_failure(conditioning_result)
    
    def _wait_for_sample_injection(self):
        """等待用户进样"""
        self.experiment_state = 'waiting_injection'
        
        # 发送进样提醒
        if self.current_experiment['injection_reminder']:
            self._notify_user(
                "请进样", 
                "润柱完成，请注入样品并点击'已完成进样'按钮",
                priority='high',
                require_action=True,
                action_buttons=['已完成进样', '取消实验']
            )
    
    def handle_injection_completed(self):
        """处理用户进样完成通知"""
        if self.experiment_state != 'waiting_injection':
            raise ValueError("当前不在等待进样状态")
        
        self.experiment_state = 'collecting'
        self._notify_user("进样完成", "开始数据采集和样品收集")
        
        if self.current_experiment['auto_proceed_after_injection']:
            self._start_experiment_loop()
    
    def _start_experiment_loop(self):
        """启动实验主循环"""
        # 初始化收集系统
        collection_config = self._prepare_collection_config()
        self.hardware.collection_module.configure(collection_config)
        
        # 启动检测器
        self.hardware.detector.start_acquisition()
        
        # 启动流程控制
        for method_id in self.current_experiment['selected_methods']:
            self._execute_method(method_id)
        
        # 实验完成
        self._experiment_completed()
    
    def _execute_method(self, method_id):
        """执行单个方法"""
        method = self.method_manager.get_method(method_id)
        assigned_tubes = self.current_experiment['method_tube_assignments'][method_id]
        
        # 设置收集试管范围
        self.hardware.collection_module.set_active_tubes(assigned_tubes)
        
        # 运行方法
        method_result = self.hardware.pump.run_method(method)
        
        return method_result
    
    def _prepare_collection_config(self):
        """准备收集配置"""
        if self.current_experiment.get('waste_mode', False):
            # 废弃模式配置
            return {
                'mode': 'waste',
                'waste_container_id': self.current_experiment['waste_container_id'],
                'track_volume': True,
                'enable_tube_collection': False
            }
        else:
            # 正常收集模式配置
            return {
                'mode': 'collection',
                'start_tube': self.current_experiment['start_tube_number'],
                'end_tube': self.current_experiment['end_tube_number'],
                'tube_assignments': self.current_experiment['method_tube_assignments'],
                'enable_tube_collection': True
            }
    
    def _switch_to_waste_container(self, waste_container_id):
        """切换液体流向到废液桶"""
        try:
            # 获取废液桶信息
            waste_container = self.db.fetchone("""
                SELECT valve_position, container_name 
                FROM waste_containers 
                WHERE id = %s AND status = 'active'
            """, (waste_container_id,))
            
            if not waste_container:
                raise ValueError("废液桶不可用")
            
            # 切换阀门到废液桶位置
            valve_position = waste_container['valve_position']
            success = self.hardware.valve_controller.switch_to_waste(valve_position)
            
            if success:
                self._record_liquid_flow(
                    source_type='pump',
                    destination_type='waste',
                    destination_waste_id=waste_container_id,
                    experiment_phase=self.experiment_state
                )
                
                self._notify_user("流向切换", f"液体已切换至废液桶: {waste_container['container_name']}")
                return True
            else:
                raise RuntimeError("阀门切换失败")
                
        except Exception as e:
            self._notify_user("错误", f"切换废液桶失败: {str(e)}", priority='error')
            return False
    
    def _record_liquid_flow(self, source_type, destination_type, 
                           destination_tube_number=None, destination_waste_id=None, 
                           experiment_phase=None, flow_rate_ml_min=None, 
                           estimated_volume_ml=None):
        """记录液体流向"""
        try:
            # 获取当前阀门配置
            valve_config = self.hardware.valve_controller.get_current_configuration()
            
            # 记录流向数据
            self.db.execute("""
                INSERT INTO liquid_flow_records 
                (experiment_id, flow_timestamp, source_type, destination_type,
                 destination_tube_number, destination_waste_id, flow_rate_ml_min,
                 estimated_volume_ml, valve_configuration, experiment_phase)
                VALUES (%s, NOW(), %s, %s, %s, %s, %s, %s, %s, %s)
            """, (
                self.current_experiment.get('experiment_id'),
                source_type, destination_type,
                destination_tube_number, destination_waste_id,
                flow_rate_ml_min, estimated_volume_ml,
                json.dumps(valve_config), experiment_phase
            ))
            
            # 更新废液桶容量（如果流向废液桶）
            if destination_type == 'waste' and destination_waste_id and estimated_volume_ml:
                self.db.execute("""
                    UPDATE waste_containers 
                    SET current_volume_ml = current_volume_ml + %s
                    WHERE id = %s
                """, (estimated_volume_ml, destination_waste_id))
                
        except Exception as e:
            print(f"记录液体流向失败: {e}")
    
    def _experiment_completed(self):
        """实验完成处理"""
        self.experiment_state = 'awaiting_user_decisions'
        
        # 停止数据采集
        self.hardware.detector.stop_acquisition()
        
        # 获取收集的馏分信息
        collected_fractions = self.hardware.collection_module.get_collected_fractions()
        
        # 通知用户做出保留/废弃决定
        if self.current_experiment['fraction_decision_required']:
            self._request_fraction_decisions(collected_fractions)
    
    def _request_fraction_decisions(self, fractions):
        """请求用户对馏分做出决定"""
        self._notify_user(
            "实验完成", 
            f"共收集 {len(fractions)} 个馏分，请决定保留或废弃",
            priority='high',
            require_action=True,
            data={'fractions': fractions}
        )
    
    def handle_fraction_decision(self, tube_number, decision):
        """处理用户的馏分决定"""
        # 更新试管状态
        self.db.execute("""
            UPDATE tube_status 
            SET user_decision = %s, decision_time = NOW(), 
                current_status = %s
            WHERE tube_number = %s AND experiment_id = %s
        """, (decision, 
              'discarded' if decision == 'discard' else 'collected',
              tube_number, 
              self.current_experiment['id']))
        
        # 检查是否所有试管都已决定
        if self._all_tubes_decided():
            self._finalize_experiment()
    
    def _all_tubes_decided(self):
        """检查是否所有试管都已做出决定"""
        pending_count = self.db.execute("""
            SELECT COUNT(*) FROM tube_status 
            WHERE experiment_id = %s AND user_decision IS NULL
        """, (self.current_experiment['id'],)).fetchone()[0]
        
        return pending_count == 0
    
    def _finalize_experiment(self):
        """完成实验最终处理"""
        # 执行废弃操作
        discarded_tubes = self.db.execute("""
            SELECT tube_number FROM tube_status 
            WHERE experiment_id = %s AND user_decision = 'discard'
        """, (self.current_experiment['id'],)).fetchall()
        
        if discarded_tubes:
            self._execute_tube_disposal([t[0] for t in discarded_tubes])
        
        # 复位系统
        self._reset_system()
        
        # 更新实验状态为完成
        self.db.execute("""
            UPDATE pre_experiment_configs 
            SET config_status = 'completed'
            WHERE id = %s
        """, (self.current_experiment['id'],))
        
        self.experiment_state = 'idle'
        self._notify_user("实验完成", "所有试管已处理完毕，系统已复位")
    
    def pause_experiment(self):
        """暂停实验"""
        if self.experiment_state not in ['collecting', 'conditioning']:
            raise ValueError(f"当前状态 {self.experiment_state} 无法暂停")
        
        # 暂停高压泵工作
        self.hardware.pump.pause()
        
        # 暂停数据积分
        self.hardware.detector.pause_integration()
        
        # 记录暂停时间
        self.current_experiment['paused_at'] = time.time()
        self.experiment_state = 'paused'
        
        self._notify_user("实验暂停", "高压泵和数据积分已暂停")
        
        return {'success': True, 'message': '实验已暂停'}
    
    def resume_experiment(self):
        """继续实验"""
        if self.experiment_state != 'paused':
            raise ValueError("当前不在暂停状态")
        
        # 继续高压泵工作
        self.hardware.pump.resume()
        
        # 继续数据积分
        self.hardware.detector.resume_integration()
        
        # 更新暂停时长记录
        if 'paused_at' in self.current_experiment:
            pause_duration = time.time() - self.current_experiment['paused_at']
            self.current_experiment.setdefault('total_pause_time', 0)
            self.current_experiment['total_pause_time'] += pause_duration
            del self.current_experiment['paused_at']
        
        self.experiment_state = 'collecting'
        
        self._notify_user("实验继续", "高压泵和数据积分已恢复")
        
        return {'success': True, 'message': '实验已继续'}
    
    def terminate_experiment(self):
        """终止实验"""
        if self.experiment_state in ['idle', 'completed']:
            raise ValueError(f"当前状态 {self.experiment_state} 无法终止")
        
        # 停止高压泵工作
        self.hardware.pump.stop()
        
        # 停止数据积分和采集
        self.hardware.detector.stop_integration()
        self.hardware.detector.stop_acquisition()
        
        # 保存当前数据
        self._save_partial_experiment_data()
        
        # 更新实验状态
        self.experiment_state = 'terminated'
        
        self._notify_user("实验终止", "实验已被用户终止，数据已保存")
        
        return {'success': True, 'message': '实验已终止'}
```

```python
class RealTimeGradientController:
    def __init__(self, pump_controller, method_manager):
        self.pump = pump_controller
        self.method_manager = method_manager
        self.current_method = None
        self.gradient_modifications = []
        
    def get_current_gradient_status(self):
        """获取当前梯度状态信息"""
        if not self.current_method:
            return {'error': '当前无运行中的方法'}
        
        current_time = self.pump.get_current_runtime()  # 获取当前运行时间(秒)
        
        if self.current_method['method_type'] == 'gradient':
            # 标准梯度方法
            current_gradient = self._get_current_gradient_point(current_time)
            return {
                'method_type': 'gradient',
                'current_time_seconds': current_time,
                'current_gradient': current_gradient,
                'modifiable': True
            }
        elif self.current_method['method_type'] == 'auto_gradient':
            # 自动梯度方法
            auto_status = self._get_auto_gradient_status(current_time)
            return {
                'method_type': 'auto_gradient',
                'current_time_seconds': current_time,
                'current_gradient': auto_status['gradient'],
                'gradient_rate': auto_status['rate'],
                'manual_hold': auto_status['manual_hold'],
                'modifiable': True
            }
        else:
            # 等度方法
            return {
                'method_type': 'isocratic',
                'current_time_seconds': current_time,
                'modifiable': False
            }
    
    def modify_gradient_point(self, new_solvent_b_percent, apply_immediately=True):
        """修改当前时间点的梯度比例（标准梯度方法）"""
        if self.current_method['method_type'] != 'gradient':
            raise ValueError("当前方法不支持梯度修改")
        
        current_time = self.pump.get_current_runtime()
        
        # 验证比例范围
        if not 0 <= new_solvent_b_percent <= 100:
            raise ValueError("溶剂B比例必须在0-100%之间")
        
        # 记录修改
        modification = {
            'timestamp': time.time(),
            'runtime_seconds': current_time,
            'old_gradient': self._get_current_gradient_point(current_time),
            'new_solvent_b_percent': new_solvent_b_percent,
            'solvent_a_percent': 100 - new_solvent_b_percent
        }
        
        self.gradient_modifications.append(modification)
        
        if apply_immediately:
            # 立即应用修改
            result = self.pump.set_immediate_gradient(
                solvent_a_percent=100 - new_solvent_b_percent,
                solvent_b_percent=new_solvent_b_percent
            )
            
            # 更新方法中的梯度程序
            self._update_method_gradient_program(current_time, new_solvent_b_percent)
            
            return {
                'success': result['success'],
                'message': f'梯度已修改为 A:{100-new_solvent_b_percent}% B:{new_solvent_b_percent}%',
                'current_time': current_time,
                'new_gradient': {
                    'solvent_a_percent': 100 - new_solvent_b_percent,
                    'solvent_b_percent': new_solvent_b_percent
                }
            }
        
        return {'success': True, 'message': '梯度修改已记录，等待应用'}
    
    def modify_auto_gradient(self, new_solvent_b_percent=None, new_gradient_rate=None, manual_hold=None):
        """修改自动梯度参数"""
        if self.current_method['method_type'] != 'auto_gradient':
            raise ValueError("当前方法不是自动梯度方法")
        
        current_time = self.pump.get_current_runtime()
        modifications = {}
        
        # 修改当前比例
        if new_solvent_b_percent is not None:
            if not 0 <= new_solvent_b_percent <= 100:
                raise ValueError("溶剂B比例必须在0-100%之间")
            modifications['solvent_b_percent'] = new_solvent_b_percent
        
        # 修改梯度速率
        if new_gradient_rate is not None:
            if not 0.1 <= new_gradient_rate <= 10.0:
                raise ValueError("梯度速率必须在0.1-10.0之间")
            modifications['gradient_rate'] = new_gradient_rate
        
        # 修改手动保持状态
        if manual_hold is not None:
            modifications['manual_hold'] = manual_hold
        
        # 应用到自动梯度算法
        if modifications:
            result = self._apply_auto_gradient_modifications(modifications)
            
            # 记录修改
            modification_record = {
                'timestamp': time.time(),
                'runtime_seconds': current_time,
                'modifications': modifications,
                'success': result['success']
            }
            self.gradient_modifications.append(modification_record)
            
            return result
        
        return {'success': False, 'message': '未提供有效的修改参数'}
    
    def _apply_auto_gradient_modifications(self, modifications):
        """应用自动梯度修改到算法"""
        try:
            # 获取当前自动梯度算法实例（从findpeaks.py）
            auto_gradient = self.pump.get_auto_gradient_controller()
            
            if 'manual_hold' in modifications:
                # 设置手动保持状态
                auto_gradient.set_manual_hold(modifications['manual_hold'])
            
            if 'solvent_b_percent' in modifications:
                # 立即设置当前比例
                result = self.pump.set_immediate_gradient(
                    solvent_a_percent=100 - modifications['solvent_b_percent'],
                    solvent_b_percent=modifications['solvent_b_percent']
                )
                if not result['success']:
                    return result
                
                # 更新算法的当前状态
                auto_gradient.update_current_gradient(modifications['solvent_b_percent'])
            
            if 'gradient_rate' in modifications:
                # 修改梯度速率
                auto_gradient.set_gradient_rate(modifications['gradient_rate'])
            
            message_parts = []
            if 'solvent_b_percent' in modifications:
                message_parts.append(f"当前比例: B {modifications['solvent_b_percent']}%")
            if 'gradient_rate' in modifications:
                message_parts.append(f"梯度速率: {modifications['gradient_rate']}")
            if 'manual_hold' in modifications:
                hold_status = "启用" if modifications['manual_hold'] else "禁用"
                message_parts.append(f"手动保持: {hold_status}")
            
            return {
                'success': True,
                'message': f'自动梯度已修改: {", ".join(message_parts)}'
            }
            
        except Exception as e:
            return {
                'success': False,
                'error': f'修改自动梯度失败: {str(e)}'
            }
    
    def _get_current_gradient_point(self, current_time):
        """获取当前时间点的梯度比例"""
        gradient_program = self.current_method.get('gradient_program', [])
        
        # 查找当前时间对应的梯度点
        for i, point in enumerate(gradient_program):
            if current_time <= point['time_point'] * 60:  # 转换分钟为秒
                if i == 0:
                    return {
                        'solvent_a_percent': point['solvent_a_percent'],
                        'solvent_b_percent': point['solvent_b_percent']
                    }
                else:
                    # 线性插值计算当前比例
                    prev_point = gradient_program[i-1]
                    time_ratio = ((current_time / 60) - prev_point['time_point']) / \
                                (point['time_point'] - prev_point['time_point'])
                    
                    current_b = prev_point['solvent_b_percent'] + \
                               (point['solvent_b_percent'] - prev_point['solvent_b_percent']) * time_ratio
                    
                    return {
                        'solvent_a_percent': 100 - current_b,
                        'solvent_b_percent': current_b
                    }
        
        # 如果超过最后一个点，返回最后一个点的值
        last_point = gradient_program[-1]
        return {
            'solvent_a_percent': last_point['solvent_a_percent'],
            'solvent_b_percent': last_point['solvent_b_percent']
        }
    
    def _get_auto_gradient_status(self, current_time):
        """获取自动梯度当前状态"""
        try:
            auto_gradient = self.pump.get_auto_gradient_controller()
            return {
                'gradient': {
                    'solvent_a_percent': auto_gradient.current_a_percent,
                    'solvent_b_percent': auto_gradient.current_b_percent
                },
                'rate': auto_gradient.current_gradient_rate,
                'manual_hold': auto_gradient.manual_hold_enabled
            }
        except Exception:
            return {
                'gradient': {'solvent_a_percent': 50, 'solvent_b_percent': 50},
                'rate': 1.0,
                'manual_hold': False
            }
    
    def _update_method_gradient_program(self, current_time, new_solvent_b_percent):
        """更新方法中的梯度程序"""
        # 找到最接近的梯度点进行更新
        gradient_program = self.current_method.get('gradient_program', [])
        current_time_min = current_time / 60
        
        for point in gradient_program:
            if abs(point['time_point'] - current_time_min) < 0.1:  # 6秒内的点
                point['solvent_b_percent'] = new_solvent_b_percent
                point['solvent_a_percent'] = 100 - new_solvent_b_percent
                break
        
        # 保存修改后的方法到数据库
        self.method_manager.update_gradient_program(
            self.current_method['id'], 
            gradient_program
        )
```

```python
class PostExperimentController:
    def __init__(self, db_connection, hardware_manager):
        self.db = db_connection
        self.hardware = hardware_manager
        self.current_experiment_id = None
        
    def set_current_experiment(self, experiment_id):

    def get_experiment_summary(self, experiment_id):
        """获取实验总结信息，包括峰和试管信息"""
        # 获取实验基本信息
        experiment = self.db.execute("""
            SELECT id, experiment_name, total_peaks, total_fractions,
                   start_time, end_time, tube_switching_timeline
            FROM experiments WHERE id = %s
        """, (experiment_id,)).fetchone()
        
        # 获取峰信息
        peaks = self.db.execute("""
            SELECT peak_id, peak_number, retention_time, peak_height,
                   peak_area, peak_start_time, peak_end_time, tube_ids
            FROM peaks WHERE experiment_id = %s
            ORDER BY peak_number
        """, (experiment_id,)).fetchall()
        
        # 获取试管信息
        fractions = self.db.execute("""
            SELECT tube_id, tube_number, switch_in_time, switch_out_time,
                   actual_volume, status, peak_ids
            FROM fractions WHERE experiment_id = %s
            ORDER BY tube_number
        """, (experiment_id,)).fetchall()
        
        # 构建峰与试管的关联关系
        peak_tube_mapping = self._build_peak_tube_mapping(peaks, fractions)
        
        return {
            'experiment': experiment,
            'peaks': peaks,
            'fractions': fractions,
            'peak_tube_mapping': peak_tube_mapping,
            'available_tubes': [f['tube_number'] for f in fractions]
        }
    
    def create_tube_operation_task(self, experiment_id, selected_tubes, operation_type, 
                                 peak_references=None, task_name=None):
        """创建试管操作任务"""
        
        # 验证试管是否属于当前实验
        experiment_tubes = self._get_experiment_tubes(experiment_id)
        invalid_tubes = [t for t in selected_tubes if t not in experiment_tubes]
        if invalid_tubes:
            raise ValueError(f"试管 {invalid_tubes} 不属于当前实验")
        
        # 验证操作类型
        if operation_type not in ['keep', 'discard']:
            raise ValueError("操作类型必须是 'keep' 或 'discard'")
        
        # 生成任务名称
        if not task_name:
            tube_list = ', '.join(map(str, selected_tubes))
            action = "保留" if operation_type == 'keep' else "废弃"
            task_name = f"{action}试管 {tube_list}"
        
        # 创建任务
        task_id = self.db.execute("""
            INSERT INTO post_experiment_tasks
            (experiment_id, task_name, task_type, selected_tubes, 
             operation_type, peak_references, created_by)
            VALUES (%s, %s, 'tube_operation', %s, %s, %s, %s)
            RETURNING id
        """, (
            experiment_id, task_name, selected_tubes, 
            operation_type, peak_references or [], 
            self._get_current_user_id()
        )).fetchone()[0]
        
        return task_id
    
    def create_batch_task_group(self, experiment_id, task_ids, batch_name=None):
        """创建批处理任务组"""
        if not task_ids:
            raise ValueError("任务列表不能为空")
        
        # 验证所有任务都属于同一个实验
        task_experiments = self.db.execute("""
            SELECT DISTINCT experiment_id FROM post_experiment_tasks
            WHERE id = ANY(%s)
        """, (task_ids,)).fetchall()
        
        if len(task_experiments) != 1 or task_experiments[0][0] != experiment_id:
            raise ValueError("所有任务必须属于同一个实验")
        
        # 生成批次名称
        if not batch_name:
            batch_name = f"批处理任务组 {len(task_ids)}个任务"
        
        # 创建批处理组
        batch_id = self.db.execute("""
            INSERT INTO task_batch_groups
            (experiment_id, batch_name, task_ids, total_tasks, created_by)
            VALUES (%s, %s, %s, %s, %s)
            RETURNING id
        """, (
            experiment_id, batch_name, task_ids, 
            len(task_ids), self._get_current_user_id()
        )).fetchone()[0]
        
        return batch_id
    
    def execute_single_task(self, task_id):
        """执行单个任务"""
        task = self._get_task(task_id)
        
        if task['status'] != 'pending':
            raise ValueError(f"任务状态为 {task['status']}，无法执行")
        
        # 更新任务状态为执行中
        self.db.execute("""
            UPDATE post_experiment_tasks 
            SET status = 'executing', started_at = NOW()
            WHERE id = %s
        """, (task_id,))
        
        try:
            if task['task_type'] == 'tube_operation':
                result = self._execute_tube_operation(task)
            elif task['task_type'] == 'system_reset':
                result = self._execute_system_reset(task)
            elif task['task_type'] == 'cleaning':
                result = self._execute_cleaning_task(task)
            else:
                raise ValueError(f"未知的任务类型: {task['task_type']}")
            
            # 更新任务完成状态
            self.db.execute("""
                UPDATE post_experiment_tasks 
                SET status = %s, completed_at = NOW(),
                    success_tubes = %s, failed_tubes = %s, error_messages = %s
                WHERE id = %s
            """, (
                'completed' if result['success'] else 'failed',
                result.get('success_tubes', []),
                result.get('failed_tubes', []),
                json.dumps(result.get('errors', {})),
                task_id
            ))
            
            return result
            
        except Exception as e:
            # 更新任务失败状态
            self.db.execute("""
                UPDATE post_experiment_tasks 
                SET status = 'failed', completed_at = NOW(),
                    error_messages = %s
                WHERE id = %s
            """, (json.dumps({'error': str(e)}), task_id))
            
            return {'success': False, 'error': str(e)}
    
    def execute_batch_tasks(self, batch_id):
        """执行批处理任务组"""
        batch = self._get_batch(batch_id)
        
        if batch['status'] != 'pending':
            raise ValueError(f"批处理任务组状态为 {batch['status']}，无法执行")
        
        # 更新批处理状态
        self.db.execute("""
            UPDATE task_batch_groups 
            SET status = 'executing', started_at = NOW()
            WHERE id = %s
        """, (batch_id,))
        
        results = []
        completed_count = 0
        failed_count = 0
        
        try:
            for task_id in batch['task_ids']:
                try:
                    result = self.execute_single_task(task_id)
                    results.append({'task_id': task_id, 'result': result})
                    
                    if result['success']:
                        completed_count += 1
                    else:
                        failed_count += 1
                        
                        # 如果设置了出错停止，则停止执行
                        if not batch.get('continue_on_error', False):
                            break
                            
                except Exception as e:
                    failed_count += 1
                    results.append({
                        'task_id': task_id, 
                        'result': {'success': False, 'error': str(e)}
                    })
                    
                    if not batch.get('continue_on_error', False):
                        break
            
            # 确定最终状态
            if failed_count == 0:
                final_status = 'completed'
            elif completed_count == 0:
                final_status = 'failed'
            else:
                final_status = 'partial_failed'
            
            # 更新批处理状态
            self.db.execute("""
                UPDATE task_batch_groups 
                SET status = %s, completed_at = NOW(),
                    completed_tasks = %s, failed_tasks = %s
                WHERE id = %s
            """, (final_status, completed_count, failed_count, batch_id))
            
            return {
                'success': failed_count == 0,
                'total_tasks': len(batch['task_ids']),
                'completed_tasks': completed_count,
                'failed_tasks': failed_count,
                'results': results
            }
            
        except Exception as e:
            # 更新批处理失败状态
            self.db.execute("""
                UPDATE task_batch_groups 
                SET status = 'failed', completed_at = NOW()
                WHERE id = %s
            """, (batch_id,))
            
            return {'success': False, 'error': str(e)}
    
    def _execute_tube_operation(self, task):
        """执行试管操作"""
        operation_type = task['operation_type']
        selected_tubes = task['selected_tubes']
        
        success_tubes = []
        failed_tubes = []
        errors = {}
        
        for tube_number in selected_tubes:
            try:
                if operation_type == 'keep':
                    # 保留试管：更新状态为已收集
                    self._update_tube_status(tube_number, 'collected', 'keep')
                    success_tubes.append(tube_number)
                    
                elif operation_type == 'discard':
                    # 废弃试管：执行物理废弃操作
                    discard_result = self.hardware.collection_module.discard_tube(tube_number)
                    if discard_result['success']:
                        self._update_tube_status(tube_number, 'discarded', 'discard')
                        success_tubes.append(tube_number)
                    else:
                        failed_tubes.append(tube_number)
                        errors[str(tube_number)] = discard_result.get('error', '废弃失败')
                        
            except Exception as e:
                failed_tubes.append(tube_number)
                errors[str(tube_number)] = str(e)
        
        return {

                json.dumps({'task_id': task['id']}),
                ['detector', 'pump', 'collection_module', 'ui'],

            )).fetchone()[0]
            
            # 执行系统复位操作
            reset_results = []
            
            # 1. 停止数据采集
            detector_result = self.hardware.detector.reset()
            reset_results.append(('detector', detector_result))
            
            # 2. 复位泵系统

            reset_results.append(('pump', pump_result))
            
            # 3. 复位收集模块

            reset_results.append(('collection_module', collection_result))
            
            # 4. 清空UI显示数据
            ui_result = self._clear_ui_data()
            reset_results.append(('ui', ui_result))
            
            # 检查是否所有操作都成功
            all_success = all(result['success'] for _, result in reset_results)
            
            # 更新系统操作记录
            self.db.execute("""
                UPDATE system_operations 
                SET status = %s, success = %s, completed_at = NOW(),
                    result_message = %s
                WHERE id = %s
            """, (
                'completed' if all_success else 'failed',
                all_success,
                json.dumps(dict(reset_results)),
                operation_id
            ))
            
            return {
                'success': all_success,
                'message': '系统复位完成' if all_success else '系统复位部分失败',
                'details': dict(reset_results)
            }
            
        except Exception as e:
            return {
                'success': False,
                'error': f'系统复位失败: {str(e)}'
```

```python
    def execute_cleaning_task(self, task_id):
        """执行清洗任务"""
        task = self._get_task(task_id)
        
        # 更新任务状态
        self.db.execute("""
            UPDATE cleaning_tasks 
            SET status = 'running', started_at = NOW()
            WHERE id = %s
        """, (task_id,))
        
        try:
            for i, tube_number in enumerate(task['tube_numbers']):
                # 更新当前清洗试管
                self.db.execute("""
                    UPDATE cleaning_tasks 
                    SET current_tube = %s, progress = %s
                    WHERE id = %s
                """, (tube_number, (i / len(task['tube_numbers'])) * 100, task_id))
                
                # 执行单个试管清洗
                self._clean_single_tube(tube_number, task)
                
                # 更新试管状态
                self.db.execute("""
                    UPDATE tube_status 
                    SET current_status = 'ready', last_cleaned_time = NOW(),
                        content_volume = 0, content_type = 'empty',
                        cleaning_required = false
                    WHERE tube_number = %s
                """, (tube_number,))
            
            # 任务完成
            self.db.execute("""
                UPDATE cleaning_tasks 
                SET status = 'completed', completed_at = NOW(), progress = 100
                WHERE id = %s
            """, (task_id,))
            
            return {'success': True, 'message': '清洗任务完成'}
            
        except Exception as e:
            # 任务失败
            self.db.execute("""
                UPDATE cleaning_tasks 
                SET status = 'failed'
                WHERE id = %s
            """, (task_id,))
            
            return {'success': False, 'error': str(e)}
    
    def _clean_single_tube(self, tube_number, task_config):
        """清洗单个试管"""
        # 移动到试管位置
        self.hardware.collection_module.move_to_position(tube_number)
        
        # 执行清洗循环
        for cycle in range(task_config['cleaning_cycles']):
            # 计算清洗体积和时间
            volume = task_config['cleaning_volume_per_tube']
            self.spray_pump.calculate_cleaning_time(volume)
            
            # 执行清洗
            result = self.spray_pump.write_info_True()
            
            if not result:
                raise Exception(f"试管 {tube_number} 清洗失败")
    
    def check_cleaning_completion(self, tube_numbers):
        """检查清洗完成状态"""
        completed = self.db.execute("""
            SELECT tube_number FROM tube_status 
            WHERE tube_number = ANY(%s) AND cleaning_required = false
        """, (tube_numbers,)).fetchall()
        
        completed_tubes = [t[0] for t in completed]
        remaining_tubes = [t for t in tube_numbers if t not in completed_tubes]
        
        return {
            'completed': len(remaining_tubes) == 0,
            'completed_tubes': completed_tubes,
            'remaining_tubes': remaining_tubes
        }
```

#### 10.2.1 实验前准备SOP

1. **系统自检**
   - 检查所有设备连接状态
   - 验证气泡传感器（气1-气4）液体状态
   - 检查废液桶液位
   - 测试UV检测器通信
   - 验证高压泵和阀门状态

2. **试管和方法配置**
   - 选择起始和结束试管号（默认全部）
   - 配置每个方法使用的试管范围
   - 设置润柱和进样选项
   - 检查需要清洗的试管

3. **清洗前置检查**
   - 如果有试管需要清洗，必须先完成清洗
   - 清洗未完成不可开始实验
   - 验证所有选中试管状态为"就绪"

#### 10.2.2 实验执行SOP

1. **润柱阶段**（可选）
   - 如果启用润柱，系统自动执行平衡
   - 监控压力和流速稳定性

2. **进样阶段**（可选）
   - 润柱完成后提醒用户进样
   - 用户手动注入样品
   - 确认进样完成后自动继续

3. **收集阶段**
   - 自动执行所选方法
   - 实时监控UV检测和试管切换
   - 记录每个试管的收集数据

4. **后处理阶段**
   - 实验完成后提醒用户决定馏分处理
   - 对每个试管选择"保留"或"废弃"
   - 自动执行废弃操作和系统复位
